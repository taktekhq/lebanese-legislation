{{t "data.example_2"}}

SELECT l.title, count(DISTINCT r.from_id) AS rulings
FROM reference r JOIN legislation l ON l.id = r.law_id
JOIN ruling x ON x.id = r.from_id AND r.from_type = 'ruling'
WHERE x.court LIKE 'تمييز%'
GROUP BY l.title ORDER BY rulings DESC LIMIT 20;

{{t "data.example_3"}}

SELECT a.title AS repealing, b.title AS repealed
FROM reference r
JOIN legislation a ON a.id = r.from_id
JOIN legislation b ON b.id = r.law_id
WHERE r.from_type = 'legislation' AND r.relation = 'repeals';
